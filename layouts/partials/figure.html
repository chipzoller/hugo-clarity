{{/* If Hugo has been able to access the image, it's loaded at .image, and we have access to additional attributes. If Hugo hasn't been able to access the image, it's loaded at .file, with only its path available. We set the Hugo variables to strings to allow for logical operations and replacements. */}} {{- $file := .file -}} {{- $image := .image -}} {{- $svgContent := .svgContent | default "" -}} {{- $diskPath := .diskPath -}} {{- $webPath := .webPath -}} {{- $alt := .alt -}} {{- $cap := .cap -}} {{- $classes := .classes -}} {{- $bundle := .bundle -}} {{/* For SVG images (which Hugo cannot decode as raster resources to provide .Width/.Height), read the . */}} {{- $svgWidth := "" -}} {{- $svgHeight := "" -}} {{- if and (not $image) (or $svgContent (in $classes "image_svg") (strings.HasSuffix (lower $file) ".svg")) -}} {{- if not (in $classes "image_svg") -}} {{- $classes = printf "%s image_svg" $classes -}} {{- end -}} {{- if and (not $svgContent) (not (strings.HasPrefix $file "http")) -}} {{- $bundleSvgPath := path.Join "content" $diskPath $file -}} {{- $staticSvgPath := path.Join "static" $file -}} {{- if and $diskPath (not (strings.HasPrefix $file "/")) (fileExists $bundleSvgPath) -}} {{- $svgContent = readFile $bundleSvgPath -}} {{- else if fileExists $staticSvgPath -}} {{- $svgContent = readFile $staticSvgPath -}} {{- end -}} {{- end -}} {{- with $svgContent -}} {{- range first 1 (findRE "(?s)]*>" . 1) -}} {{- $wMatch := findRESubmatch `\swidth=["']\s*(\d+(?:\.\d+)?)\s*(?:px)?\s*["']` . 1 -}} {{- $hMatch := findRESubmatch `\sheight=["']\s*(\d+(?:\.\d+)?)\s*(?:px)?\s*["']` . 1 -}} {{- if $wMatch }}{{ $svgWidth = printf "%.0f" (math.Round (float (index (index $wMatch 0) 1))) }}{{ end -}} {{- if $hMatch }}{{ $svgHeight = printf "%.0f" (math.Round (float (index (index $hMatch 0) 1))) }}{{ end -}} {{- if or (not $svgWidth) (not $svgHeight) -}} {{- $vbMatch := findRESubmatch `\sviewBox=["']\s*[\d.-]+[\s,]+[\d.-]+[\s,]+(\d+(?:\.\d+)?)[\s,]+(\d+(?:\.\d+)?)\s*["']` . 1 -}} {{- if $vbMatch -}} {{- $svgWidth = printf "%.0f" (math.Round (float (index (index $vbMatch 0) 1))) -}} {{- $svgHeight = printf "%.0f" (math.Round (float (index (index $vbMatch 0) 2))) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $fileWeb := .file -}} {{- if eq $bundle true -}} {{- $fileWeb = path.Join $webPath $file -}} {{- if or (strings.HasPrefix $file "http") (strings.HasPrefix $file "/") -}} {{- $fileWeb = $file -}} {{- end -}} {{- end -}}
